#   pip install torch --index-url https://download.pytorch.org/whl/cpu
#   pip install -r requirements-dev.txt
# The suite never downloads models and never touches CUDA. Keep this file ASCII-only
# (pip decodes it with the locale code page on Windows).
pytest

# 5.x. The suite passes on both; tests build script UIs inside gr.Blocks, as Forge does.
gradio<6.0,>=4.0
pydantic
numpy
Pillow>=11.1.0
fastapi
packaging
opencv-python-headless

scipy
PyWavelets
# gradio -> pydub imports audioop, which Python 3.13 removed (Forge installs this too).
audioop-lts; python_version >= "3.13"
